%%%%%%   プログラムを付け加えて質問を行なう。

  &program;;
  &rule;;
  花田勝/[勝数=14]<=花田勝;;

X = 横綱 if { 花田勝.勝数==_V1, _V1#>#13 }

    % 〔次の番付は横綱〕
